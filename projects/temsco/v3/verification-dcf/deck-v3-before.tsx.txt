function Metric({ label, value, unit = '억 원', detail, negative = false }: { label: string; value: string; unit?: string; detail: string; negative?: boolean }) {
  return <div className={s.metric}><p className={s.metricLabel}>{label}</p><div className={`${s.metricValue} ${negative ? s.negative : ''}`}>{value}<small>{unit}</small></div><p className={s.metricDetail}>{detail}</p></div>
}
function Editorial({ rows }: { rows: [string, string][] }) {
  return <div className={s.editorial}>{rows.map(([title, body], i) => <div key={title} className={s.editorialRow}><span>{String(i + 1).padStart(2, '0')}</span><div><h3>{title}</h3><p>{body}</p></div></div>)}</div>
}
function Matrix({ heads, rows }: { heads: string[]; rows: ReactNode[][] }) {
  return <table className={`${s.table} ${s.matrix}`}><thead><tr>{heads.map(h => <th key={h} scope="col">{h}</th>)}</tr></thead><tbody>{rows.map((row, i) => <tr key={i}>{row.map((cell, j) => <td key={j}>{cell}</td>)}</tr>)}</tbody></table>
}
